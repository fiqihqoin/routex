# Go Integration

Example of integration using the standard `net/http` library in Go.

## Code Example

<CodeBlock language="go" filename="routex.go">
{`package main

import (
    "bytes"
    "encoding/json"
    "net/http"
)

func createQRIS(orderId string, amount int) {
    url := "https://sandbox.routex.id/api/v1/transactions"
    payload := map[string]interface{}{
        "amount": amount,
        "currency": "IDR",
        "payment_channel": "qris",
    }
    
    jsonPayload, _ := json.Marshal(payload)
    req, _ := http.NewRequest("POST", url, bytes.NewBuffer(jsonPayload))
    
    req.Header.Set("X-API-Key", "your_key")
    req.Header.Set("X-Idempotency-Key", orderId)
    req.Header.Set("Content-Type", "application/json")
    
    client := &http.Client{}
    resp, _ := client.Do(req)
    defer resp.Body.Close()
}`}
</CodeBlock>
